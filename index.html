// HOW IT WORKS

256× Faster with Bitmap

In ERC721A, the ownerOf(tokenId) call scans backwards by reading a separate storage slot for each token — in a batch of 1000 tokens, this means 1000 SLOADs.

UltraERC721 skips 256 tokens at once in each bucket. Thanks to Bitmap, only a few bucket reads are needed. A real difference.

/// ownerOf — 256x faster than ERC721A function ownerOf(uint256 tokenId) public view returns (address) { if (tokenId >= totalSupply) revert OwnerQueryForNonexistentToken(); // Fast path: explicit owner (batch head or transferred) address owner = _owners[tokenId]; if (owner != address(0)) return owner; // Slow path: bucket scanning with bitmap — each bucket 256 tokens! uint256 bucket = tokenId >> 8; // tokenId / 256 uint256 bitPos = tokenId & 0xff; // tokenId % 256 if (bitPos > 0) { uint256 mask = (1 << bitPos) - 1; uint256 bucketData = _batchHead._data[bucket] & mask; if (bucketData != 0) { uint256 msb = _highestBitSet(bucketData); return _owners[(bucket << 8) | msb]; } } // Look at previous buckets while (bucket > 0) { unchecked { bucket--; } uint256 bucketData = _batchHead._data[bucket]; if (bucketData != 0) return _owners[(bucket << 8) | _highestBitSet(bucketData)]; } revert OwnerQueryForNonexistentToken(); }
/// Batch mint — only 1 storage write for 1000 tokens! function _mint(address to, uint256 quantity) internal virtual { if (to == address(0)) revert MintToZeroAddress(); if (quantity == 0) revert MintZeroQuantity(); uint256 startTokenId = totalSupply; unchecked { _balances[to] += quantity; totalSupply += quantity; } // Lazy init: only FIRST token of BATCH writes to storage _owners[startTokenId] = to; _batchHead.set(startTokenId); // Emit event as per ERC721 standard for (uint256 i = 0; i < quantity; ) { emit Transfer(address(0), to, startTokenId + i); unchecked { ++i; } } }
/// Find highest set bit in O(1) time /// Branch-free binary search — 8 operations, no loops function _highestBitSet(uint256 x) internal pure returns (uint256 r) { require(x > 0); assembly { r := shl(7, lt(0xffffffffffffffffffffffffffffffff, x)) r := or(r, shl(6, lt(0xffffffffffffffff, shr(r, x)))) r := or(r, shl(5, lt(0xffffffff, shr(r, x)))) r := or(r, shl(4, lt(0xffff, shr(r, x)))) r := or(r, shl(3, lt(0xff, shr(r, x)))) r := or(r, shl(2, lt(0xf, shr(r, x)))) r := or(r, shl(1, lt(0x3, shr(r, x)))) r := or(r, lt(0x1, shr(r, x))) } // Result: position of highest set bit in x (0-indexed) // Usage: to find batch head in bucket scanning }
